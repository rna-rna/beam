// Modify the handleDrag function to update in real-time:
const handleDrag = (clientX: number, clientY: number) => {
  if (!isDragging || !containerRef.current) return;

  // Get container dimensions
  const rect = containerRef.current.getBoundingClientRect();

  // Calculate new percentage position
  const newX = ((clientX - rect.left) / rect.width) * 100;
  const newY = ((clientY - rect.top) / rect.height) * 100;

  // Clamp values between 0 and 100
  const boundedX = Math.max(0, Math.min(100, newX));
  const boundedY = Math.max(0, Math.min(100, newY));

  // Update local position state immediately for visual feedback
  setPosition({ x: boundedX, y: boundedY });
  
  // Call onPositionChange continuously during drag for real-time updates
  if (onPositionChange) {
    // Use a debounced version to prevent too many server updates
    onPositionChange(boundedX, boundedY);
  }
};
